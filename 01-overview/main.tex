\section{程序的执行过程}

\begin{frame}{从源代码到可执行文件}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.46\textwidth}
      \begin{itemize}
        \item 编译器把源文件翻译成目标文件。
        \item 链接器把目标文件和用到的库合并成可执行文件。
        \item 操作系统装载可执行文件，创建进程并开始执行。
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[
        node distance=9mm,
        box/.style={
          draw, rounded corners, align=center,
          inner sep=4pt, minimum width=0.85\linewidth
        },
        ar/.style={-{Latex[length=2mm]}, thick}
      ]
        \node[box] (src) {源代码\\\texttt{main.c}};
        \node[box, below=of src] (obj) {目标文件\\\texttt{main.o}};
        \node[box, below=of obj] (exe) {可执行文件\\\texttt{a.out}};
        \draw[ar] (src) -- node[right, font=\footnotesize] {编译器} (obj);
        \draw[ar] (obj) -- node[right, font=\footnotesize] {链接器} (exe);
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{三个阶段的分工}
  \centering
  \small
  \begin{tabular}{@{}lll@{}}
    \toprule
    阶段 & 负责者 & 主要工作 \\
    \midrule
    编译期 & 编译器 & 检查语法、生成目标代码 \\
    链接期 & 链接器 & 解析符号、合并库函数 \\
    运行期 & 操作系统 & 装载程序、分配内存、调度进程 \\
    \bottomrule
  \end{tabular}
  \medskip
  \par
  \normalsize
  一次编译并不等于程序已经可以运行。
\end{frame}

\section{代码示例}

\begin{frame}[fragile]{函数返回值如何传回调用方}
  \begin{lstlisting}[language=Python]
def square(value):
    return value * value

result = square(4)
print(result)  # 16
  \end{lstlisting}
  \medskip
  \texttt{return} 把计算结果交回调用方，调用方再用变量接收。
\end{frame}

\begin{frame}{小结}
  \begin{vfilleditems}
    \item 源代码需要经过编译、链接才能成为可执行文件。
    \item 编译期、链接期和运行期由不同角色负责。
    \item 理解的入口是“每一步的输入和输出是什么”。
  \end{vfilleditems}
\end{frame}
