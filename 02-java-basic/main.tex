\begin{frame}[fragile]{让程序主动“开口”}
  我们需要知道程序正在执行什么，可以让它把信息打印到控制台。
  \par\medskip
  使用 \texttt{System.out.println(...)} 输出一行文字：
  \begin{lstlisting}[language=Java]
System.out.println("Hello, Java!");
  \end{lstlisting}
  双引号里面的文字会被打印出来，双引号本身不会显示。
  \par\medskip
  控制台输出：
  \begin{lstlisting}[language={}]
Hello, Java!
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{把输出语句写在 main 中}
  程序从 \texttt{main} 方法开始执行。先把输出语句写在它的大括号内：
  \begin{lstlisting}[language=Java]
public class Main {
    public static void main(String[] args) {
        System.out.println("First");
        System.out.println("Second");
    }
}
  \end{lstlisting}
  这里的两条语句\textbf{从上到下依次执行}，每次输出后换行：
  \begin{lstlisting}[language={}]
First
Second
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{注释：给人看的说明}
  注释用于解释代码，注释中的内容不会被当作语句执行。
  \begin{itemize}
    \item 单行注释：\texttt{//} 后面直到本行末尾的内容。
    \item 多行注释：\texttt{/*} 与 \texttt{*/} 之间的内容，可以跨行。
  \end{itemize}
  \medskip
  下面的代码写在 \texttt{main} 中：
  \begin{lstlisting}[language=Java]
// Print a greeting.
System.out.println("Hello!"); // This is a comment.
/* The next statement is commented out.
System.out.println("Hidden");
*/
System.out.println("Goodbye!");
  \end{lstlisting}
  控制台依次输出 \texttt{Hello!} 和 \texttt{Goodbye!}，各占一行；
  \texttt{Hidden} 不会输出。
\end{frame}

\begin{frame}[fragile]{变量：保存程序需要的数据}
  程序有时需要做运算，也需要保存一些数据，比如数值、文本。
  \par\medskip
  可以用\textbf{变量}保存这些数据，再通过变量名读取或修改它们。
  \par\medskip
  下面的代码写在 \texttt{main} 中：
  \begin{lstlisting}[language=Java]
int score = 60;
System.out.println(score);

score = 70;
System.out.println(score);
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{int score = 60;}：声明整数变量 \texttt{score}，初始化为 60。
    \item \texttt{score = 70;}：重新赋值，把 \texttt{score} 中保存的值改为 70。
    \item 两次 \texttt{println} 依次输出 60 和 70，各占一行。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{变量的声明与赋值}
  \textbf{声明变量时赋值（初始化）}
  \par
  语法：\quad 类型\quad 变量名\quad \texttt{=}\quad 初始值\texttt{;}
  \begin{lstlisting}[language=Java]
int score = 60;
  \end{lstlisting}
  \medskip
  \textbf{也可以先声明，再赋值}
  \par
  声明：\quad 类型\quad 变量名\texttt{;}
  \par
  赋值：\quad 变量名\quad \texttt{=}\quad 值\texttt{;}
  \begin{lstlisting}[language=Java]
int score;
score = 60;
System.out.println(score);
  \end{lstlisting}
  赋值时不用再写类型。
\end{frame}

% 类型范围依据 Java Language Specification §4.2：
% https://docs.oracle.com/javase/specs/jls/se25/html/jls-4.html#jls-4.2
\begin{frame}[fragile]{整数类型：能保存的范围不同}
  \texttt{byte}、\texttt{short}、\texttt{int} 和 \texttt{long} 都能保存整数。
  \par\medskip
  它们占用的位数不同，能保存的整数范围也不同：
  \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{lcll}
      \textbf{类型} & \textbf{位数} & \textbf{取值范围} & \textbf{声明并赋值} \\
      \hline
      \texttt{byte} & 8 & $-128 \sim 127$ & \texttt{byte level = 10;} \\
      \texttt{short} & 16 & $-32768 \sim 32767$ & \texttt{short count = 1000;} \\
      \texttt{int} & 32 & $-2^{31} \sim 2^{31}-1$ & \texttt{int score = 60;} \\
      \texttt{long} & 64 & $-2^{63} \sim 2^{63}-1$ & \texttt{long total = 3000000000L;} \\
    \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item 一般使用 \texttt{int} 即可，范围约为负 21 亿到正 21 亿。
    \item 需要更大范围时使用 \texttt{long}；上例数字末尾的 \texttt{L} 表示 \texttt{long}。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{浮点类型：保存带小数的数值}
  \texttt{float} 和 \texttt{double} 的取值范围与精度不同。
  \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{lclc}
      \textbf{类型} & \textbf{位数} & \textbf{有限值范围（约）} & \textbf{有效数字} \\
      \hline
      \texttt{float} & 32 & $-3.4\times10^{38} \sim 3.4\times10^{38}$ & 约 7 位 \\
      \texttt{double} & 64 & $-1.8\times10^{308} \sim 1.8\times10^{308}$ & 约 15--16 位 \\
    \end{tabular}
  \end{center}
  \begin{lstlisting}[language=Java]
float temperature = 36.5F;
double height = 1.75;
  \end{lstlisting}
  \begin{itemize}
    \item 一般使用 \texttt{double}；带小数点的数值默认是 \texttt{double}。
    \item 给 \texttt{float} 赋值时，上例在数字末尾加 \texttt{F}。
    \item 浮点数通常是近似值，不能精确表示范围内的每一个小数。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{布尔类型：只有 true 和 false}
  \texttt{boolean} 只有两个值：\texttt{true}（真）和 \texttt{false}（假）。
  \par\medskip
  \begin{lstlisting}[language=Java]
boolean ready = true;
System.out.println(ready);

ready = false;
System.out.println(ready);
  \end{lstlisting}
  两次输出依次为 \texttt{true} 和 \texttt{false}，各占一行。
  \par\medskip
  先记住它的写法，具体作用留到后面介绍。
\end{frame}

\begin{frame}[fragile]{字符类型：用单引号表示一个字符}
  \texttt{char} 可以保存一个英文字母，比如 \texttt{A}。
  \par\medskip
  \begin{lstlisting}[language=Java]
char letter = 'A';
System.out.println(letter);

letter = 'B';
System.out.println(letter);
  \end{lstlisting}
  \begin{itemize}
    \item 字符使用\textbf{单引号}，如 \texttt{'A'}；不要写成双引号 \texttt{"A"}。
    \item 两次输出依次为 \texttt{A} 和 \texttt{B}，引号本身不会显示。
  \end{itemize}
\end{frame}

% 转换规则：https://docs.oracle.com/javase/specs/jls/se25/html/jls-5.html
\begin{frame}[fragile]{自动转换与强制转换}
  \textbf{自动转换}：如 \texttt{int} 转为 \texttt{long} 或 \texttt{double}，Java 会自动完成。
  \begin{lstlisting}[language=Java]
int count = 10;
double value = count;
System.out.println(value); // 10.0
  \end{lstlisting}
  \textbf{强制转换}：浮点数转整数，或整数变量转为范围更小的类型时，
  需要显式写出 \texttt{(目标类型)}。
  \begin{lstlisting}[language=Java]
double price = 19.9;
int whole = (int) price;
System.out.println(whole); // 19
  \end{lstlisting}
  \begin{itemize}
    \item 上例转为整数时舍去小数部分，不是四舍五入。
    \item 强制转换可能丢失信息；自动转换也不一定保留全部精度。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{算术运算符：加、减、乘、除与取余}
  我们可以对数字做运算，把结果保存到变量中，或直接打印出来。
  \begin{center}
    \renewcommand{\arraystretch}{1.15}
    \begin{tabular}{cllc}
      \textbf{符号} & \textbf{运算} & \textbf{表达式示例} & \textbf{结果} \\
      \hline
      \texttt{+} & 加法 & \texttt{7 + 2} & 9 \\
      \texttt{-} & 减法 & \texttt{7 - 2} & 5 \\
      \texttt{*} & 乘法 & \texttt{7 * 2} & 14 \\
      \texttt{/} & 除法 & \texttt{7 / 2} & 3 \\
      \texttt{\%} & 取余 & \texttt{7 \% 2} & 1 \\
    \end{tabular}
  \end{center}
  \begin{lstlisting}[language=Java]
int sum = 7 + 2;
System.out.println(sum);   // 9
  \end{lstlisting}
  \texttt{\%} 求除法的余数，常被称为“取模”；这里先看正整数。
\end{frame}

\begin{frame}[fragile]{括号可以改变运算顺序}
  默认先计算乘法、除法和取余，再计算加法、减法。
  \par\medskip
  \textbf{没有括号：先乘，再加}
  \begin{lstlisting}[language=Java]
int result = 2 + 3 * 4;
System.out.println(result); // 14
  \end{lstlisting}
  先算 \texttt{3 * 4} 得到 12，再算 \texttt{2 + 12} 得到 14。
  \par\medskip
  \textbf{加上括号：先计算括号里面的部分}
  \begin{lstlisting}[language=Java]
result = (2 + 3) * 4;
System.out.println(result); // 20
  \end{lstlisting}
  先算 \texttt{2 + 3} 得到 5，再算 \texttt{5 * 4} 得到 20。
\end{frame}

\begin{frame}[fragile]{超出整数范围：结果可能变成负数}
  \texttt{byte} 的范围是 $-128 \sim 127$，无法保存 128。
  \begin{lstlisting}[language=Java]
byte number = 127;
System.out.println(number + 1); // 128
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{byte} 参与这里的加法时，先提升为 \texttt{int}，得到 128。
    \item 强制转回 \texttt{byte} 时只保留低 8 位，结果变为 $-128$。
    \item 转换不会自动检查范围或报错；应选择范围足够大的类型。
  \end{itemize}
  \medskip
  \texttt{short} 对应的边界是 32767：加 1 后转回 \texttt{short} 得到 $-32768$。
\end{frame}

% 整数除法：https://docs.oracle.com/javase/specs/jls/se25/html/jls-15.html#jls-15.17.2
\begin{frame}[fragile]{整数除法：先转换，再相除}
  两个整数相除，结果仍是整数，小数部分向零截断。
  \begin{lstlisting}[language=Java]
int total = 7;
int count = 2;
System.out.println(total / count);         // 3
System.out.println((double) total / count); // 3.5
  \end{lstlisting}
  想保留小数，先把其中一个操作数转成 \texttt{double}，再做除法。
  \par\medskip
  \textbf{只把结果赋给 double，或除完再转换，都来不及：}
  \begin{lstlisting}[language=Java]
double result = total / count;
System.out.println(result);                // 3.0
System.out.println((double) (total / count)); // 3.0
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{自增、自减与运算后赋值}
  \textbf{自增与自减}：\texttt{++} 让变量加 1，\texttt{--} 让变量减 1。
  \begin{lstlisting}[language=Java]
int count = 5;
count++; // count is now 6
count--; // count is now 5
  \end{lstlisting}
  \textbf{复合赋值}：把运算结果保存回原变量。以下以 \texttt{int} 变量为例：
  \begin{center}
    \renewcommand{\arraystretch}{1.1}
    \begin{tabular}{ll}
      \textbf{简写} & \textbf{对应写法} \\
      \hline
      \texttt{count += 2;} & \texttt{count = count + 2;} \\
      \texttt{count -= 2;} & \texttt{count = count - 2;} \\
      \texttt{count *= 2;} & \texttt{count = count * 2;} \\
      \texttt{count /= 2;} & \texttt{count = count / 2;} \\
      \texttt{count \%= 2;} & \texttt{count = count \% 2;} \\
    \end{tabular}
  \end{center}
\end{frame}

\begin{frame}[fragile]{比较运算：结果是 boolean}
  比较两个数值，结果是 \texttt{true} 或 \texttt{false}，可以保存到 \texttt{boolean} 变量中。
  \begin{center}
    \renewcommand{\arraystretch}{1.05}
    \begin{tabular}{cllc}
      \textbf{运算符} & \textbf{含义} & \textbf{示例} & \textbf{结果} \\
      \hline
      \texttt{==} & 等于 & \texttt{7 == 2} & \texttt{false} \\
      \texttt{!=} & 不等于 & \texttt{7 != 2} & \texttt{true} \\
      \texttt{>} & 大于 & \texttt{7 > 2} & \texttt{true} \\
      \texttt{<} & 小于 & \texttt{7 < 2} & \texttt{false} \\
      \texttt{>=} & 大于或等于 & \texttt{7 >= 7} & \texttt{true} \\
      \texttt{<=} & 小于或等于 & \texttt{7 <= 2} & \texttt{false} \\
    \end{tabular}
  \end{center}
  \begin{lstlisting}[language=Java]
boolean passed = 80 >= 60;
System.out.println(passed); // true
  \end{lstlisting}
  注意：\texttt{=} 是赋值，\texttt{==} 才是判断是否相等。
\end{frame}

\begin{frame}[fragile]{逻辑运算：组合或取反条件}
  逻辑运算的操作数和结果都是 \texttt{boolean}。
  \begin{center}
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{cll}
      \textbf{运算符} & \textbf{含义} & \textbf{结果为 true 的条件} \\
      \hline
      \texttt{\&\&} & 与 & 两边都为 \texttt{true} \\
      \texttt{||} & 或 & 至少一边为 \texttt{true} \\
      \texttt{!} & 非（取反） & 原来的值为 \texttt{false} \\
    \end{tabular}
  \end{center}
  \begin{lstlisting}[language=Java]
int score = 80;
boolean inRange = (score >= 0) && (score <= 100); // true
boolean outside = (score < 0) || (score > 100);   // false
boolean failed = !(score >= 60);                // false
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{字符串：保存文本信息}
  字符串可以理解为一串 \texttt{char}，用来保存文本信息。
  \par\medskip
  使用 \texttt{String} 声明变量，用\textbf{双引号}包住文字：
  \begin{lstlisting}[language=Java]
String name = "Alice";
name = "Bob";
System.out.println(name); // Bob
  \end{lstlisting}
  \texttt{String} 的首字母要大写，它不属于八种基本数据类型。
  \par\medskip
  \textbf{有无双引号，含义不同：}
  \begin{lstlisting}[language=Java]
int age = 18;
System.out.println(age);   // 18
System.out.println("age"); // age
  \end{lstlisting}
  \texttt{age} 读取变量的值；\texttt{"age"} 表示文字本身。
\end{frame}

\begin{frame}[fragile]{用 + 拼接字符串}
  \texttt{+} 可以把字符串连接起来；数值参与拼接时会转换为文本。
  \begin{lstlisting}[language=Java]
String name = "Alice";
int age = 18;
String message = name + " is " + age + " years old.";
System.out.println(message); // Alice is 18 years old.
  \end{lstlisting}
  \textbf{注意区分数值加法和字符串拼接：}
  \begin{lstlisting}[language=Java]
System.out.println(1 + 2);           // 3
System.out.println("Total: " + 1 + 2);   // Total: 12
System.out.println("Total: " + (1 + 2)); // Total: 3
  \end{lstlisting}
  \begin{itemize}
    \item 连续的 \texttt{+} 从左向右组合，遇到字符串就进行拼接。
    \item 想先算出数值结果，再拼接文字，就把计算部分放在括号中。
  \end{itemize}
\end{frame}

\begin{frame}{综合示例：商品总价计算需求}
  购买 A、B 两种商品，根据各自的单价和数量计算总价。
  \par\bigskip
  \begin{itemize}
    \item 用 \texttt{priceA}、\texttt{priceB} 保存两种商品的单价，单位为元。
    \item 用 \texttt{quantityA}、\texttt{quantityB} 保存各自的购买数量。
    \item 直接修改代码中四个变量的初始值。
    \item 计算总价，并在控制台打印结果。
  \end{itemize}
  \bigskip
  \textbf{计算公式：}总价 = A 的单价 $\times$ 数量 + B 的单价 $\times$ 数量。
  \par\medskip
  示例数据：A 每件 12.5 元，买 2 件；B 每件 8 元，买 3 件。
\end{frame}

\begin{frame}[fragile]{综合示例：商品总价计算实现}
  \begin{lstlisting}[language=Java]
public class Main {
    public static void main(String[] args) {
        double priceA = 12.5;
        double priceB = 8.0;
        int quantityA = 2;
        int quantityB = 3;
        double total = priceA * quantityA + priceB * quantityB;
        System.out.println("Total: " + total);
    }
}
  \end{lstlisting}
  \begin{itemize}
    \item 先分别计算两种商品的金额，再相加得到总价。
    \item 本例输出 \texttt{Total: 49.0}，即总价为 49 元。
  \end{itemize}
\end{frame}

\begin{frame}{新需求：满足条件时才打折}
  现在增加一个需求：总价\textbf{超过 40 元}时打八折。
  \par
  也就是：当 \texttt{total > 40} 时，让 \texttt{total} 乘以 0.8；否则保持不变。
\end{frame}

\begin{frame}[fragile]{if：满足条件时才执行}
  \textbf{if 语法：条件为 true 时执行，否则跳过。}
  \begin{lstlisting}[language=Java,escapeinside={(*@}{@*)}]
if ((*@条件@*)) {
    (*@条件成立时执行的语句@*)
}
  \end{lstlisting}
  条件必须是 \texttt{boolean}。例如：总价超过 40 元时打八折。
  \begin{lstlisting}[language=Java]
double total = 49.0;
if (total > 40) {
    total = total * 0.8;
}
System.out.println("Total: " + total); // Total: 39.2
  \end{lstlisting}
\end{frame}

\begin{frame}{新需求：打折与未打折时输出不同文字}
  如果打折了，输出文字也要变化：
  \begin{itemize}
    \item 打折时，输出 \texttt{Discounted total:} 和折后总价。
    \item 未打折时，输出 \texttt{Total:} 和正常总价。
  \end{itemize}
  \bigskip
  原来的写法只在 \texttt{if} 中修改价格，最后总会执行外面的同一条输出语句，
  因而无法区分这两种文字。
  \par\bigskip
  自然的想法是：\textbf{如果打折了，就按折扣文本打印；否则，按正常文本打印。}
  \par\medskip
  用 \texttt{else} 表示“否则”，让程序在两条分支中选择一条执行。
\end{frame}

\begin{frame}[fragile]{if--else：在两条分支中选择一条}
  条件为 \texttt{true} 时执行 \texttt{if} 分支，为 \texttt{false} 时执行 \texttt{else} 分支。
  \par\medskip
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.35\textwidth}
      \textbf{语法}
  \begin{lstlisting}[language=Java,escapeinside={(*@}{@*)}]
if ((*@条件@*)) {
    (*@条件成立时执行@*)
} else {
    (*@条件不成立时执行@*)
}
  \end{lstlisting}
    \end{column}
    \begin{column}{0.62\textwidth}
      \textbf{折扣示例}
  \begin{lstlisting}[language=Java]
double total = 49.0;
if (total > 40) {
    total = total * 0.8;
    System.out.println(
        "Discounted total: " + total);
} else {
    System.out.println(
        "Total: " + total);
}
  \end{lstlisting}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{新需求：按总价划分价格档次}
  假设总价非负，我们希望根据金额输出对应的价格档次。
  \par\bigskip
  \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{ll}
      \textbf{总价范围} & \textbf{档次} \\
      \hline
      $0 \leq \texttt{total} \leq 40$ & 便宜（\texttt{Cheap}） \\
      $40 < \texttt{total} \leq 100$ & 适中（\texttt{Moderate}） \\
      $\texttt{total} > 100$ & 昂贵（\texttt{Expensive}） \\
    \end{tabular}
  \end{center}
  \bigskip
  40 元属于“便宜”，100 元属于“适中”，每个总价只属于一个档次。
\end{frame}

\begin{frame}[fragile]{先用三个独立的 if 实现}
  \begin{lstlisting}[language=Java]
double total = 80.0;
if (total <= 40) {
    System.out.println("Cheap");
}
if (total > 40 && total <= 100) {
    System.out.println("Moderate");
}
if (total > 100) {
    System.out.println("Expensive");
}
  \end{lstlisting}
  这三个判断在逻辑上是一组：对\textbf{同一个不变的总价}，恰好只有一个条件成立。
  \par\medskip
  但三个独立的 \texttt{if} 仍会从上到下，逐个判断条件。
\end{frame}

\begin{frame}[fragile]{修改总价后，后面的 if 可能再次成立}
  假设超过 100 元打五折，并把这个判断放在另外两个判断之间：
  \begin{lstlisting}[language=Java]
double total = 180.0;
if (total <= 40) {
    System.out.println("Cheap");
}
if (total > 100) {
    total = total * 0.5;
    System.out.println("Discounted");
}
if (total > 40 && total <= 100) {
    System.out.println("Moderate");
}
  \end{lstlisting}
  180 元变成 90 元后，又满足“适中”条件，两个分支都执行了。
\end{frame}

\begin{frame}[fragile]{else if：一组条件，只执行一个分支}
  我们需要把多个判断连成一组：\textbf{一旦某个条件成立，后面的条件就不再判断。}
  \par\medskip
  \begin{lstlisting}[language=Java,escapeinside={(*@}{@*)}]
if ((*@条件 1@*)) {
    (*@条件 1 成立时执行的语句@*)
} else if ((*@条件 2@*)) {
    (*@条件 2 成立时执行的语句@*)
} else {
    (*@以上条件都不成立时执行的语句@*)
}
  \end{lstlisting}
  \begin{itemize}
    \item 从上到下判断，只执行\textbf{第一个成立条件}对应的分支。
    \item 即使分支中修改了变量，也不会继续判断后面的 \texttt{else if}。
    \item 最后的 \texttt{else} 可以省略；省略后可能一个分支也不执行。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{新需求：余额不足时充值}
  商品总价 180 元，余额只有 50 元。假设每次充值增加 50 元。
  \par\medskip
  先用 \texttt{if}：余额不足，就充值一次。
  \begin{lstlisting}[language=Java]
double total = 180.0;
double balance = 50.0;
if (balance < total) {
    balance = balance + 50;
}
System.out.println("Balance: " + balance); // 100.0
  \end{lstlisting}
  \textbf{问题：充值一次后，余额仍然不够付款。}
  \par\medskip
  \texttt{if} 只判断一次，不会自动回头检查余额是否已经足够。
\end{frame}

\begin{frame}{充值后重新检查，直到余额足够}
  我们需要的是：\textbf{只要余额不足，就继续充值。}
  \par\bigskip
  \begin{enumerate}
    \item 检查余额是否小于商品总价。
    \item 如果不足，充值 50 元，然后回到第一步。
    \item 如果已经足够，结束充值，继续付款。
  \end{enumerate}
  \bigskip
  本例余额变化：$50 \rightarrow 100 \rightarrow 150 \rightarrow 200$ 元。
  \par\medskip
  余额达到 200 元时，不再满足“余额不足”，于是停止充值。
  \par\medskip
  用 \texttt{while} 表示这种\textbf{条件成立就反复执行}的过程。
\end{frame}

\begin{frame}[fragile]{while：条件成立就重复执行}
  每轮先判断条件：为 \texttt{true} 时执行循环体，再回到条件处；为 \texttt{false} 时退出。
  \par\medskip
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.33\textwidth}
      \textbf{语法}
      \begin{lstlisting}[language=Java,escapeinside={(*@}{@*)}]
while ((*@条件@*)) {
    (*@反复执行的语句@*)
}
      \end{lstlisting}
      条件必须是 \texttt{boolean}。
    \end{column}
    \begin{column}{0.64\textwidth}
      \textbf{充值与付款}
      \begin{lstlisting}[language=Java]
double total = 180.0;
double balance = 50.0;
while (balance < total) {
    balance = balance + 50;
}
balance = balance - total;
System.out.println(
    "Remaining: " + balance); // 20.0
      \end{lstlisting}
    \end{column}
  \end{columns}
  \medskip
  \begin{itemize}
    \item 一开始余额就足够时，循环一次也不执行，直接付款。
    \item 如果余额不足，却不在循环中增加余额，就会一直循环。
  \end{itemize}
\end{frame}
