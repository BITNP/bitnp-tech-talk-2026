\begin{frame}[fragile]{让程序主动“开口”}
  我们需要知道程序正在执行什么，可以让它把信息打印到控制台。
  \par\medskip
  使用 \texttt{System.out.println(...)} 输出一行文字：
  \begin{lstlisting}[language=Java]
System.out.println("Hello, Java!");
  \end{lstlisting}
  双引号里面的文字会被打印出来，双引号本身不会显示。
  \par\medskip
  控制台输出：
  \begin{lstlisting}[language={}]
Hello, Java!
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{把输出语句写在 main 中}
  程序从 \texttt{main} 方法开始执行。先把输出语句写在它的大括号内：
  \begin{lstlisting}[language=Java]
public class Main {
    public static void main(String[] args) {
        System.out.println("First");
        System.out.println("Second");
    }
}
  \end{lstlisting}
  这里的两条语句\textbf{从上到下依次执行}，每次输出后换行：
  \begin{lstlisting}[language={}]
First
Second
  \end{lstlisting}
\end{frame}

\begin{frame}[fragile]{注释：给人看的说明}
  注释用于解释代码，注释中的内容不会被当作语句执行。
  \begin{itemize}
    \item 单行注释：\texttt{//} 后面直到本行末尾的内容。
    \item 多行注释：\texttt{/*} 与 \texttt{*/} 之间的内容，可以跨行。
  \end{itemize}
  \medskip
  下面的代码写在 \texttt{main} 中：
  \begin{lstlisting}[language=Java]
// Print a greeting.
System.out.println("Hello!"); // This is a comment.
/* The next statement is commented out.
System.out.println("Hidden");
*/
System.out.println("Goodbye!");
  \end{lstlisting}
  控制台依次输出 \texttt{Hello!} 和 \texttt{Goodbye!}，各占一行；
  \texttt{Hidden} 不会输出。
\end{frame}

\begin{frame}[fragile]{变量：保存程序需要的数据}
  程序有时需要做运算，也需要保存一些数据，比如数值、文本。
  \par\medskip
  可以用\textbf{变量}保存这些数据，再通过变量名读取或修改它们。
  \par\medskip
  下面的代码写在 \texttt{main} 中：
  \begin{lstlisting}[language=Java]
int score = 60;
System.out.println(score);

score = 70;
System.out.println(score);
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{int score = 60;}：声明整数变量 \texttt{score}，初始化为 60。
    \item \texttt{score = 70;}：重新赋值，把 \texttt{score} 中保存的值改为 70。
    \item 两次 \texttt{println} 依次输出 60 和 70，各占一行。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{变量的声明与赋值}
  \textbf{声明变量时赋值（初始化）}
  \par
  语法：\quad 类型\quad 变量名\quad \texttt{=}\quad 初始值\texttt{;}
  \begin{lstlisting}[language=Java]
int score = 60;
  \end{lstlisting}
  \medskip
  \textbf{也可以先声明，再赋值}
  \par
  声明：\quad 类型\quad 变量名\texttt{;}
  \par
  赋值：\quad 变量名\quad \texttt{=}\quad 值\texttt{;}
  \begin{lstlisting}[language=Java]
int score;
score = 60;
System.out.println(score);
  \end{lstlisting}
  赋值时不用再写类型。
\end{frame}

\begin{frame}[fragile]{整数类型：能保存的范围不同}
  \texttt{byte}、\texttt{short}、\texttt{int} 和 \texttt{long} 都能保存整数。
  \par\medskip
  它们占用的位数不同，能保存的整数范围也不同：
  \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{lcll}
      \textbf{类型} & \textbf{位数} & \textbf{取值范围} & \textbf{声明并赋值} \\
      \hline
      \texttt{byte} & 8 & $-128 \sim 127$ & \texttt{byte level = 10;} \\
      \texttt{short} & 16 & $-32768 \sim 32767$ & \texttt{short count = 1000;} \\
      \texttt{int} & 32 & $-2^{31} \sim 2^{31}-1$ & \texttt{int score = 60;} \\
      \texttt{long} & 64 & $-2^{63} \sim 2^{63}-1$ & \texttt{long total = 3000000000L;} \\
    \end{tabular}
  \end{center}
  \medskip
  \begin{itemize}
    \item 一般使用 \texttt{int} 即可，范围约为负 21 亿到正 21 亿。
    \item 需要更大范围时使用 \texttt{long}；上例数字末尾的 \texttt{L} 表示 \texttt{long}。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{浮点类型：保存带小数的数值}
  \texttt{float} 和 \texttt{double} 的取值范围与精度不同。
  \begin{center}
    \renewcommand{\arraystretch}{1.4}
    \begin{tabular}{lclc}
      \textbf{类型} & \textbf{位数} & \textbf{有限值范围（约）} & \textbf{有效数字} \\
      \hline
      \texttt{float} & 32 & $-3.4\times10^{38} \sim 3.4\times10^{38}$ & 约 7 位 \\
      \texttt{double} & 64 & $-1.8\times10^{308} \sim 1.8\times10^{308}$ & 约 15--16 位 \\
    \end{tabular}
  \end{center}
  \begin{lstlisting}[language=Java]
float temperature = 36.5F;
double height = 1.75;
  \end{lstlisting}
  \begin{itemize}
    \item 一般使用 \texttt{double}；带小数点的数值默认是 \texttt{double}。
    \item 给 \texttt{float} 赋值时，上例在数字末尾加 \texttt{F}。
    \item 浮点数通常是近似值，不能精确表示范围内的每一个小数。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{布尔类型：只有 true 和 false}
  \texttt{boolean} 只有两个值：\texttt{true}（真）和 \texttt{false}（假）。
  \par\medskip
  \begin{lstlisting}[language=Java]
boolean ready = true;
System.out.println(ready);

ready = false;
System.out.println(ready);
  \end{lstlisting}
  两次输出依次为 \texttt{true} 和 \texttt{false}，各占一行。
  \par\medskip
  先记住它的写法，具体作用留到后面介绍。
\end{frame}

\begin{frame}[fragile]{字符类型：用单引号表示一个字符}
  \texttt{char} 可以保存一个英文字母，比如 \texttt{A}。
  \par\medskip
  \begin{lstlisting}[language=Java]
char letter = 'A';
System.out.println(letter);

letter = 'B';
System.out.println(letter);
  \end{lstlisting}
  \begin{itemize}
    \item 字符使用\textbf{单引号}，如 \texttt{'A'}；不要写成双引号 \texttt{"A"}。
    \item 两次输出依次为 \texttt{A} 和 \texttt{B}，引号本身不会显示。
  \end{itemize}
\end{frame}

% 转换规则：https://docs.oracle.com/javase/specs/jls/se25/html/jls-5.html
\begin{frame}[fragile]{自动转换与强制转换}
  \textbf{自动转换}：如 \texttt{int} 转为 \texttt{long} 或 \texttt{double}，Java 会自动完成。
  \begin{lstlisting}[language=Java]
int count = 10;
double value = count;
System.out.println(value); // 10.0
  \end{lstlisting}
  \textbf{强制转换}：浮点数转整数，或整数变量转为范围更小的类型时，
  需要显式写出 \texttt{(目标类型)}。
  \begin{lstlisting}[language=Java]
double price = 19.9;
int whole = (int) price;
System.out.println(whole); // 19
  \end{lstlisting}
  \begin{itemize}
    \item 上例转为整数时舍去小数部分，不是四舍五入。
    \item 强制转换可能丢失信息；自动转换也不一定保留全部精度。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{算术运算符：加、减、乘、除与取余}
  我们可以对数字做运算，把结果保存到变量中，或直接打印出来。
  \begin{center}
    \renewcommand{\arraystretch}{1.15}
    \begin{tabular}{cllc}
      \textbf{符号} & \textbf{运算} & \textbf{表达式示例} & \textbf{结果} \\
      \hline
      \texttt{+} & 加法 & \texttt{7 + 2} & 9 \\
      \texttt{-} & 减法 & \texttt{7 - 2} & 5 \\
      \texttt{*} & 乘法 & \texttt{7 * 2} & 14 \\
      \texttt{/} & 除法 & \texttt{7 / 2} & 3 \\
      \texttt{\%} & 取余 & \texttt{7 \% 2} & 1 \\
    \end{tabular}
  \end{center}
  \begin{lstlisting}[language=Java]
int sum = 7 + 2;
System.out.println(sum);   // 9
  \end{lstlisting}
  \texttt{\%} 求除法的余数，常被称为“取模”；这里先看正整数。
\end{frame}

\begin{frame}[fragile]{超出整数范围：结果可能变成负数}
  \texttt{byte} 的范围是 $-128 \sim 127$，无法保存 128。
  \begin{lstlisting}[language=Java]
byte number = 127;
System.out.println(number + 1); // 128
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{byte} 参与这里的加法时，先提升为 \texttt{int}，得到 128。
    \item 强制转回 \texttt{byte} 时只保留低 8 位，结果变为 $-128$。
    \item 转换不会自动检查范围或报错；应选择范围足够大的类型。
  \end{itemize}
  \medskip
  \texttt{short} 对应的边界是 32767：加 1 后转回 \texttt{short} 得到 $-32768$。
\end{frame}

% 整数除法：https://docs.oracle.com/javase/specs/jls/se25/html/jls-15.html#jls-15.17.2
\begin{frame}[fragile]{整数除法：先转换，再相除}
  两个整数相除，结果仍是整数，小数部分向零截断。
  \begin{lstlisting}[language=Java]
int total = 7;
int count = 2;
System.out.println(total / count);         // 3
System.out.println((double) total / count); // 3.5
  \end{lstlisting}
  想保留小数，先把其中一个操作数转成 \texttt{double}，再做除法。
  \par\medskip
  \textbf{只把结果赋给 double，或除完再转换，都来不及：}
  \begin{lstlisting}[language=Java]
double result = total / count;
System.out.println(result);                // 3.0
System.out.println((double) (total / count)); // 3.0
  \end{lstlisting}
\end{frame}
