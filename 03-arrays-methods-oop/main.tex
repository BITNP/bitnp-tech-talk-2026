\begin{frame}[fragile]{三件商品：用变量保存价格}
  购买三件商品，价格分别为 12 元、8 元和 25 元，计算总价。
  \par\medskip
  下面的代码写在 \texttt{main} 中：
  \begin{lstlisting}[language=Java]
int price1 = 12;
int price2 = 8;
int price3 = 25;

int total = price1 + price2 + price3;
System.out.println(total);
  \end{lstlisting}
  \begin{itemize}
    \item 每件商品的价格保存在一个变量中。
    \item 把三个价格相加，输出总价 \texttt{45}。
  \end{itemize}
\end{frame}

\begin{frame}{商品变成三十件，怎么办}
  现在有 \textbf{30 件商品}，需要计算总价，并找出价格低于 10 元的商品。
  \par\medskip
  \begin{itemize}
    \item 保存价格：要从 \texttt{price1} 一直声明到 \texttt{price30}。
    \item 计算总价：要把 30 个变量逐个写进求和表达式。
    \item 筛选商品：要对每个价格重复写判断语句。
  \end{itemize}
  \medskip
  计算时，我们并不关心每个价格叫 \texttt{price1} 还是 \texttt{price2}，
  只需要对\alert{一组同类型的数据进行批量计算}。
  \par\medskip
  我们需要把这些价格\alert{放在一起，并通过编号访问}，这就要用到\textbf{数组}。
\end{frame}

\begin{frame}[fragile]{通过索引读取和修改数组元素}
  使用\textbf{数组名[索引]}访问指定元素；\texttt{prices[1]} 表示第二个元素。
  \par\medskip
  \begin{lstlisting}[language=Java]
int[] prices = {12, 8, 25};
System.out.println(prices[1]);  // 8
int savedPrice = prices[1];
prices[1] = 10;
System.out.println(prices[1]);  // 10
System.out.println(savedPrice); // 8
  \end{lstlisting}
  \begin{itemize}
    \item \textbf{读取}：可以直接打印元素，也可以用它参与计算。
    \item \textbf{保存}：\texttt{int savedPrice = prices[1];} 把当前的值存入新变量。
    \item \textbf{修改}：\texttt{prices[1] = 10;} 把第二个元素改为 10。
  \end{itemize}
  \medskip
  \texttt{savedPrice} 保存的是当时复制的整数值，\alert{不会随数组元素的修改而改变}。
\end{frame}

\begin{frame}[fragile]{用 for 循环计算总价}
  遍历三个价格，把每个价格累加到总价中。
  \begin{lstlisting}[language=Java]
int[] prices = {12, 8, 25};
int total = 0;
for (int i = 0; i < 3; i++) {
    total += prices[i];
}
System.out.println(total);
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{i} 依次为 0、1、2，\texttt{prices[i]} 依次取出 12、8、25。
    \item 每轮执行 \texttt{total += prices[i];}，累加当前价格。
    \item 循环结束后，打印总价 \texttt{45}。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{用 length 获取数组长度}
  \texttt{prices.length} 表示数组中元素的个数，这里是 \texttt{3}。
  \par\medskip
  把循环条件中的 \texttt{i < 3} 改为 \texttt{i < prices.length}：
  \begin{lstlisting}[language=Java]
int[] prices = {12, 8, 25};
int total = 0;
for (int i = 0; i < prices.length; i++) {
    total += prices[i];
}
System.out.println(total);
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{length} 后面没有括号。
    \item 索引从 0 到 \texttt{length - 1}，所以循环条件使用 \texttt{<}。
    \item 增减初始化列表中的价格时，不需要再修改循环条件。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{同一个公式，要重复写几遍}
  给定 \texttt{double} 类型的 $x$，计算并打印 $x^2 + 4x + 10$。
  \begin{lstlisting}[language=Java]
double x1 = 1.0;
double x2 = 2.0;
double x3 = 3.0;
System.out.println(x1 * x1 + 4 * x1 + 10);
System.out.println(x2 * x2 + 4 * x2 + 10);
System.out.println(x3 * x3 + 4 * x3 + 10);
  \end{lstlisting}
  \begin{itemize}
    \item 输入不同，计算过程相同，却要重复书写同一个公式。
    \item 如果公式中的 \texttt{+ 10} 改为 \texttt{+ 11}，每一处都要修改，容易遗漏。
  \end{itemize}
  \medskip
  可以把计算过程写成\alert{函数}，需要时调用；在 Java 中，函数通常称为\textbf{方法}。
\end{frame}

\begin{frame}[fragile]{定义方法，把计算过程写一次}
  这里先使用 \texttt{static} 方法，定义形式为：
  \par
  \texttt{static} 返回类型 方法名(参数列表) \texttt{\{} 方法体 \texttt{\}}
  \par
  方法定义在类中，与 \texttt{main} 并列：
  \begin{lstlisting}[language=Java]
static double calculate(double x) {
    double result = x * x + 4 * x + 10;
    return result;
}
  \end{lstlisting}
  在 \texttt{main} 中调用，并打印返回的结果：
  \begin{lstlisting}[language=Java]
System.out.println(calculate(1.0)); // 15.0
System.out.println(calculate(2.0)); // 22.0
System.out.println(calculate(3.0)); // 31.0
  \end{lstlisting}
  参数 \texttt{x} 接收输入；\texttt{return} 返回一个 \texttt{double} 类型的结果。
\end{frame}

\begin{frame}[fragile]{方法定义：返回类型、名字和参数列表}
  \texttt{static} 返回类型 方法名(参数列表) \texttt{\{} 方法体 \texttt{\}}
  \begin{lstlisting}[language=Java]
static double calculate(double x) {
    double result = x * x + 4 * x + 10;
    return result;
}
  \end{lstlisting}
  \begin{itemize}
    \item \textbf{返回类型}：\texttt{double} 说明返回结果的类型，不是结果本身。
    \item \textbf{方法名}：\texttt{calculate} 是调用这个方法时使用的名字。
    \item \textbf{参数列表}：每个参数写成“类型 名字”，多个参数用逗号分隔；
      没有参数时仍保留 \texttt{()}。
  \end{itemize}
  \medskip
  先把计算结果存入 \texttt{result}，再用 \texttt{return result;}\alert{结束当前方法并返回结果}。
  返回的值必须与声明的返回类型兼容。
\end{frame}

\begin{frame}[fragile]{调用方法：传入参数，使用返回值}
  调用形式：\textbf{方法名(实际参数)}，参数按定义中的顺序传入。
  \par\medskip
  下面的代码写在同一个类的 \texttt{main} 中：
  \begin{lstlisting}[language=Java]
double result = calculate(2.0);
System.out.println(result);          // 22.0
System.out.println(calculate(3.0));  // 31.0
double total = calculate(1.0) + calculate(2.0);
System.out.println(total);           // 37.0
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{calculate(2.0)} 将 \texttt{2.0} 传给参数 \texttt{x}，然后执行方法体。
    \item 执行 \texttt{return} 后回到调用处，用返回的值继续计算或赋值。
    \item 调用时不写参数类型；参数的数量、顺序和类型要与定义匹配。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{完整程序：调用后返回原处继续执行}
  保存为 \texttt{Main.java}，程序从 \texttt{main} 开始执行。
  \par\medskip
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.66\textwidth}
      \begin{lstlisting}[language=Java]
public class Main {
  public static void main(String[] args) {
    double value = calculate(2.0);
    System.out.println(value);
  }

  static double calculate(double x) {
    double result = x * x + 4 * x + 10;
    return result;
  }
}
      \end{lstlisting}
    \end{column}
    \begin{column}{0.30\textwidth}
      \textbf{执行顺序}
      \begin{enumerate}
        \item 进入 \texttt{main}。
        \item 调用 \texttt{calculate}，传入 \texttt{2.0}。
        \item 计算得到 \texttt{result = 22.0}。
        \item 返回 \texttt{main}，将结果赋给 \texttt{value}。
        \item 继续打印 \texttt{22.0}。
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{参数按位置对应，顺序不能随意交换}
  定义一个接收两个参数的方法：
  \begin{lstlisting}[language=Java]
static double sum(int a, double b) {
    double result = a + b;
    return result;
}
  \end{lstlisting}
  在 \texttt{main} 中调用：
  \begin{lstlisting}[language=Java]
double total = sum(3, 2.5);
System.out.println(total); // 5.5
  \end{lstlisting}
  \begin{itemize}
    \item 第一个实参 \texttt{3} 传给 \texttt{int a}，第二个实参 \texttt{2.5} 传给 \texttt{double b}。
    \item 实参按\alert{位置}对应形参，不会根据类型自动调整顺序。
    \item \texttt{sum(2.5, 3)} 无法编译：\texttt{2.5} 不能直接传给 \texttt{int a}。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{不同方法可以有同名的局部变量}
  下面两个方法并列定义在同一个类中：
  \par\medskip
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{lstlisting}[language=Java]
static double first() {
    double value = 10.0;
    return value;
}
      \end{lstlisting}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{lstlisting}[language=Java]
static double second() {
    double value = 20.0;
    return value;
}
      \end{lstlisting}
    \end{column}
  \end{columns}
  在 \texttt{main} 中依次调用，两个方法都能正常运行：
  \begin{lstlisting}[language=Java]
System.out.println(first());  // 10.0
System.out.println(second()); // 20.0
  \end{lstlisting}
  \begin{itemize}
    \item \textbf{作用域}是可以通过变量名访问这个变量的代码范围。
    \item 两个 \texttt{value} 是各自方法中的局部变量，名字相同也互不冲突。
    \item \texttt{main} 不能直接访问它们，但可以接收方法返回的值。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{抽象：知道怎么用，不必知道怎么实现}
  调用一个方法，需要了解它的\alert{输入、返回结果和功能约定}。
  \par\medskip
  以 \texttt{double sum(int a, double b)} 为例，约定它返回两个参数的和：
  \begin{itemize}
    \item \textbf{传入什么}：先传一个 \texttt{int}，再传一个 \texttt{double}。
    \item \textbf{得到什么}：得到一个 \texttt{double} 类型的结果，值为两者之和。
    \item \textbf{如何使用}：调用方法，再保存、打印或继续使用返回值。
  \end{itemize}
  \begin{lstlisting}[language=Java]
double total = sum(3, 2.5);
System.out.println(total); // 5.5
  \end{lstlisting}
  调用者不需要知道内部变量叫什么、计算分几步完成。
  只要实现遵守约定，就可以通过调用获得预期结果。
\end{frame}

\begin{frame}[fragile]{方法也可以调用另一个方法}
  同一个类中，\texttt{calculate} 调用 \texttt{square} 来计算平方。
  \par\medskip
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.64\textwidth}
      \begin{lstlisting}[language=Java]
static double calculate(double x) {
    double squared = square(x);
    double result = squared + 4 * x + 10;
    return result;
}

static double square(double x) {
    double result = x * x;
    return result;
}
      \end{lstlisting}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{调用与返回顺序}
      \begin{enumerate}
        \item \texttt{calculate} 接收 \texttt{x = 2.0}。
        \item 调用 \texttt{square(x)}，传入 \texttt{2.0}。
        \item \texttt{square} 返回 \texttt{4.0}，存入 \texttt{squared}。
        \item 回到 \texttt{calculate}，继续计算并返回 \texttt{22.0}。
      \end{enumerate}
    \end{column}
  \end{columns}
  每次调用都会进入被调用的方法，\alert{返回后从本次调用的位置继续执行}。
\end{frame}

\begin{frame}[fragile]{一件商品，需要保存多种信息}
  现在除了价格，还要记录每件商品的名称和库存。
  \par\medskip
  可以先用三个数组分别保存：
  \begin{lstlisting}[language=Java]
String[] names = {"Notebook", "Pen", "Eraser"};
double[] prices = {12.0, 8.0, 3.0};
int[] stocks = {10, 20, 30};
  \end{lstlisting}
  \begin{itemize}
    \item 索引 0：\texttt{Notebook}，价格 12 元，库存 10 件。
    \item 索引 1：\texttt{Pen}，价格 8 元，库存 20 件。
    \item 索引 2：\texttt{Eraser}，价格 3 元，库存 30 件。
  \end{itemize}
  \medskip
  我们约定：\alert{三个数组中相同索引的数据，属于同一件商品}。
\end{frame}

\begin{frame}[fragile]{只交换名称，商品信息就对应错了}
  想把 \texttt{Pen} 排在 \texttt{Notebook} 前面，却只交换了名称：
  \begin{lstlisting}[language=Java]
String temp = names[0];
names[0] = names[1];
names[1] = temp;
  \end{lstlisting}
  \centering
  {\renewcommand{\arraystretch}{1.3}
  \begin{tabular}{clcc}
    索引 & 名称 & 价格（元） & 库存（件） \\
    \hline
    0 & \texttt{Pen} & 12.0 & 10 \\
    1 & \texttt{Notebook} & 8.0 & 20 \\
    2 & \texttt{Eraser} & 3.0 & 30 \\
  \end{tabular}\par}
  \medskip
  \raggedright
  \texttt{Pen} 被配上了 \texttt{Notebook} 的价格和库存，但程序不会因此报错。
  \par\medskip
  调整顺序时，三个数组必须同步修改；\alert{商品信息之间的对应关系，全靠我们维护}。
\end{frame}

\begin{frame}{用类描述商品，把相关数据组织在一起}
  名称、价格和库存，本来就在描述同一件商品。
  我们希望移动一件商品时，这些信息也能\alert{作为一个整体}移动。
  \par\bigskip
  用 \texttt{Product} 类描述：\textbf{每件商品都有名称、价格和库存}。
  \par\medskip
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{一个商品对象}
        名称：\texttt{Notebook}\par
        价格：12 元\par
        库存：10 件
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{另一个商品对象}
        名称：\texttt{Pen}\par
        价格：8 元\par
        库存：20 件
      \end{block}
    \end{column}
  \end{columns}
  \bigskip
  \textbf{类}描述共同的结构；\textbf{对象（实例）}保存某一件具体商品的数据。
\end{frame}

\begin{frame}[fragile]{定义类：把相关的量写在一起}
  使用 \texttt{class 类名 \{}\,\texttt{...}\,\texttt{\}} 定义一个类。
  将下面的代码保存为 \texttt{Product.java}：
  \begin{lstlisting}[language=Java]
class Product {
    String name;
    double price;
    int stock;
}
  \end{lstlisting}
  \begin{itemize}
    \item \texttt{Product} 是类名，也可以作为变量的类型。
    \item 类中声明的这些量称为\textbf{字段（成员变量）}。
    \item 每个 \texttt{Product} 对象都有自己的 \texttt{name}、\texttt{price} 和 \texttt{stock}。
  \end{itemize}
  \medskip
  这段定义描述商品有哪些数据；具体商品的数据要保存在创建出的\alert{对象}中。
\end{frame}

\begin{frame}[fragile]{用商品对象替代三个分散的数组}
  下面的代码写在同一目录下 \texttt{Main.java} 的 \texttt{main} 中：
  \par\medskip
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{lstlisting}[language=Java]
Product notebook = new Product();
notebook.name = "Notebook";
notebook.price = 12.0;
notebook.stock = 10;
Product pen = new Product();
pen.name = "Pen";
pen.price = 8.0;
pen.stock = 20;
Product eraser = new Product();
eraser.name = "Eraser";
eraser.price = 3.0;
eraser.stock = 30;
      \end{lstlisting}
    \end{column}
    \begin{column}{0.41\textwidth}
      \begin{itemize}
        \item \texttt{new Product()} 创建一个商品对象。
        \item 用“\textbf{对象变量.字段名}”访问或修改这个对象的数据。
        \item 三件商品分别保存各自的名称、价格和库存。
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{调整商品顺序，只需交换对象}
  接着把三个商品对象放进一个 \texttt{Product[]} 数组：
  \begin{lstlisting}[language=Java]
Product[] products = {notebook, pen, eraser};

Product temp = products[0];
products[0] = products[1];
products[1] = temp;
  \end{lstlisting}
  此时，索引 0 对应的是 \texttt{pen}，它仍保存自己的价格和库存：
  \begin{lstlisting}[language=Java]
System.out.println(products[0].name);  // Pen
System.out.println(products[0].price); // 8.0
System.out.println(products[0].stock); // 20
  \end{lstlisting}
  \medskip
  每个数组元素对应\alert{一个完整的商品对象}，不必再同步交换三个数组中的数据。
\end{frame}

\begin{frame}[fragile]{构造方法：创建对象时就把数据填好}
  把字段赋值写进\textbf{构造方法}，创建对象时直接传入初始数据。
  \begin{lstlisting}[language=Java]
class Product {
    String name;
    double price;
    int stock;
    Product(String initialName, double initialPrice,
            int initialStock) {
        name = initialName;
        price = initialPrice;
        stock = initialStock;
    }
}
  \end{lstlisting}
  \begin{itemize}
    \item 构造方法与类\alert{同名}，不写返回类型，也不写 \texttt{void}。
    \item 执行 \texttt{new Product(...)} 时，会调用匹配的构造方法。
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{创建时传入参数，构造方法完成初始化}
  在 \texttt{main} 中创建商品，不再逐条给字段赋值：
  \begin{lstlisting}[language=Java]
Product notebook = new Product("Notebook", 12.0, 10);
Product pen = new Product("Pen", 8.0, 20);
Product eraser = new Product("Eraser", 3.0, 30);
  \end{lstlisting}
  参数按顺序传给构造方法，创建出的对象已经保存名称、价格和库存。
  \par\medskip
  \textbf{构造方法里也可以执行其他语句}，例如在字段赋值后打印创建提示：
  \begin{lstlisting}[language=Java]
// At the end of the constructor:
System.out.println("Created: " + name);
  \end{lstlisting}
  每次通过这个构造方法创建商品，都会执行这条提示语句。
  \par\medskip
  构造方法用于\alert{完成对象的初始化}，可以赋值，也可以进行计算或条件判断。
