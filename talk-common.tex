% talk-common.tex —— 各场分享共用的导言区配置
% 放在仓库根目录，由各个入口 .tex 通过 \input{talk-common.tex} 引用。
% 编译时需要把仓库根目录加入 TEXINPUTS，详见根目录 AGENTS.md 第 2 节。

\usetheme[customfont,showmaxslides,nofooterlogo]{pureminimalistic}
\usefonttheme{professionalfonts}
\usepackage[UTF8,fontset=none]{ctex}
\usepackage{graphicx}
\usepackage{listings}
\usepackage{appendixnumberbeamer}

\setsansfont{Fira Sans}
\setmonofont{Fira Mono}
\setCJKmainfont{Noto Sans CJK SC}
\setCJKsansfont{Noto Sans CJK SC}
\setCJKmonofont{Noto Sans CJK SC}

% 默认不引用上游示例 logo，有真实素材时再统一配置
\renewcommand{\logotitle}{}
\renewcommand{\logoheader}{}
\renewcommand{\logofooter}{}
\renewcommand{\pageword}{}

% 字号设置必须放在加载主题之后，避免被主题覆盖
\setbeamerfont{title}{size=\LARGE,shape=\upshape,series=\bfseries}
\setbeamerfont{presentation title}{shape=\upshape}
\setbeamerfont{frametitle}{size=\Large,shape=\upshape,series=\bfseries}

% 浅色配色：海军蓝标题、蓝绿强调、石墨色正文。
% 使用主题提供的接口；所有分享共用同一组语义色。
\definecolor{TalkBg}{HTML}{FFFFFF}
\definecolor{TalkText}{HTML}{253343}
\definecolor{TalkTitle}{HTML}{19354A}
\definecolor{TalkAccent}{HTML}{147D82}
\definecolor{TalkMuted}{HTML}{657382}
\definecolor{TalkSurface}{HTML}{F2F5F7}
\definecolor{TalkWarm}{HTML}{98633B}

\renewcommand{\beamerbgcolor}{TalkBg}
\renewcommand{\beamertextcolor}{TalkText}
\renewcommand{\beamertitlecolor}{TalkTitle}
\renewcommand{\beamerfootertextcolor}{TalkMuted}

% 标题与强调分工：标题沉稳，重点和列表标记使用蓝绿色。
\setbeamercolor{structure}{fg=TalkAccent}
\setbeamercolor{alerted text}{fg=TalkAccent,bg=TalkBg}
\setbeamercolor{example text}{fg=TalkAccent,bg=TalkBg}
\setbeamercolor{itemize item}{fg=TalkAccent}
\setbeamercolor{itemize subitem}{fg=TalkAccent}
\setbeamercolor{itemize subsubitem}{fg=TalkAccent}
\setbeamercolor{enumerate item}{fg=TalkAccent}
\setbeamercolor{enumerate subitem}{fg=TalkAccent}
\setbeamercolor{enumerate subsubitem}{fg=TalkAccent}
\setbeamercolor{block title}{fg=TalkTitle,bg=TalkTitle!8!TalkBg}
\setbeamercolor{block body}{fg=TalkText,bg=TalkSurface}
\setbeamercolor{block title alerted}{fg=TalkAccent,bg=TalkAccent!10!TalkBg}
\setbeamercolor{block body alerted}{fg=TalkText,bg=TalkAccent!4!TalkBg}
\setbeamercolor{block title example}{fg=TalkAccent,bg=TalkAccent!10!TalkBg}
\setbeamercolor{block body example}{fg=TalkText,bg=TalkAccent!4!TalkBg}
\hypersetup{colorlinks=true,linkcolor=TalkTitle,urlcolor=TalkAccent,citecolor=TalkAccent}

% 代码复用整套配色；暖棕仅用于字符串，注释保持足够对比度。
\colorlet{TalkCodeBg}{TalkSurface}
\colorlet{TalkCodeFg}{TalkText}
\colorlet{TalkCodeKeyword}{TalkTitle}
\definecolor{TalkCodeComment}{HTML}{63766F}
\colorlet{TalkCodeString}{TalkWarm}
\lstdefinestyle{talkcode}{
  basicstyle=\ttfamily\small\color{TalkCodeFg},
  backgroundcolor=\color{TalkCodeBg},
  keywordstyle=\bfseries\color{TalkCodeKeyword},
  commentstyle=\color{TalkCodeComment},
  stringstyle=\color{TalkCodeString},
  columns=fullflexible,
  keepspaces=true,
  showstringspaces=false,
  breaklines=true,
  tabsize=4,
  numbers=none,
  frame=none
}
\lstset{style=talkcode}

% 页脚署名开关：默认只显示“短标题 | 副标题”，需要时由本场入口追加
% \renewcommand{\talkfooterauthor}{\insertshortauthor} 才会带上短作者。
% 直接改写 footline，因为上游的版式名已被 \defbeamertemplate 占用、无法再次注册；
% 若哪天去掉主题的 nofooterlogo 选项，页脚会退回上游默认版式。
% 判空不用 \@empty：本导言区里 \@ 已被重新定义，\@empty 展开不再是空宏，
% 主题自带的 \ifx...\@empty 判断因此失效（副标题为空时会多印一个分隔符）。
\makeatletter
\newcommand{\talk@empty}{}
\newcommand{\talkfooterauthor}{}
\setbeamertemplate{footline}{%
  \basicfooter{\myfooterheight}{%
    \vbox to\myfooterheight{%
      \vfil%
      \insertshorttitle{}%
      \ifx\insertsubtitle\talk@empty%
        \else%
        ~|~ \insertshortsubtitle{}%
      \fi%
      \ifx\talkfooterauthor\talk@empty%
        \else%
        ~|~ \talkfooterauthor%
      \fi%
      \hfill \showpagenum%
      \par\vfil\vfil
    }%
  }%
}
\makeatother
